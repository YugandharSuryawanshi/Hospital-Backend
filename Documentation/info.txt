CREATE TABLE bills (
    bill_id INT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    patient_id INT UNSIGNED NOT NULL,
    reference_type ENUM(
        'appointment',
        'lab',
        'pharmacy',
        'room',
        'operation',
        'other'
    ) NOT NULL,
    reference_id INT UNSIGNED NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    tax DECIMAL(10,2) DEFAULT 0,
    discount DECIMAL(10,2) DEFAULT 0,
    final_amount DECIMAL(10,2) NOT NULL,
    bill_status ENUM('unpaid','partial','paid') DEFAULT 'unpaid',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX(patient_id),
    CONSTRAINT fk_bill_patient
        FOREIGN KEY (patient_id)
        REFERENCES users(user_id)
        ON DELETE CASCADE
) ENGINE=InnoDB;

const handleSubmit = async (e) => {
        e.preventDefault();

        if (!formData.user_name || !formData.user_contact) {
            toastError("Fill required fields");
            return;
        }

        const appointmentPayload = {
            doctor_id,
            ...formData
        };

        try {
            // OFFLINE
            if (paymentMode === "offline") {
                const res = await userAxios.post("/addAppointment", {
                    ...appointmentPayload,
                    payment_mode: "offline"
                });

                alert(`Token: ${res.data.token_number}`);
                toastSuccess(res.data.message);
                navigate("/profile");
                return;
            }

            // ONLINE
            const orderRes = await userAxios.post("/create-order", {
                doctor_id
            });

            const options = {
                key: orderRes.data.key,
                amount: orderRes.data.order.amount,
                currency: "INR",
                order_id: orderRes.data.order.id,

                handler: async function (response) {
                    const verifyRes = await userAxios.post("/verify-payment", {
                        ...response,
                        appointmentData: appointmentPayload,
                        bill_id: orderRes.data.bill_id
                    });

                    alert(`Token: ${verifyRes.data.token_number}`);
                    toastSuccess("Appointment Confirmed");
                    navigate("/profile");
                },
            };

            const rzp = new window.Razorpay(options);
            rzp.open();

        } catch (err) {
            console.log(err);
            toastError("Booking failed");
        }
    };
